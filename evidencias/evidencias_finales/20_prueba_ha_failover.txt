    maxconn 4096
    daemon

    log global
    mode tcp
    option tcplog
    timeout connect 10s
    timeout client 1m
    timeout server 1m

frontend tidb_frontend
    bind *:3390
    default_backend tidb_backend

    balance roundrobin
    option tcp-check
    server tidb1 10.0.0.174:4000 check
    server tidb2 10.0.0.73:4000 check
    server tidb3 10.0.0.158:4000 check

listen haproxy_stats
    bind *:8080
    mode http
    stats enable
    stats uri /stats
    stats refresh 5s
    stats auth admin:admin123
